cmake_minimum_required(VERSION 3.22.1)
project(karma_mnn_arm64 LANGUAGES CXX)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "KARMA MNN native backend only supports Android arm64-v8a")
endif()
set(MNN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/prebuilt" CACHE PATH "MNN Android arm64 prebuilt SDK")
if(NOT EXISTS "${MNN_ROOT}/include/MNN/Interpreter.hpp")
  message(FATAL_ERROR "Missing MNN headers: supply a real licensed arm64 MNN SDK at ${MNN_ROOT}")
endif()
if(NOT EXISTS "${MNN_ROOT}/lib/arm64-v8a/libMNN.so")
  message(FATAL_ERROR "Missing real arm64-v8a libMNN.so: do not ship a stub")
endif()
add_library(MNN SHARED IMPORTED)
set_target_properties(MNN PROPERTIES IMPORTED_LOCATION "${MNN_ROOT}/lib/arm64-v8a/libMNN.so")
add_library(karma_mnn SHARED src/karma_mnn_jni.cpp)
target_include_directories(karma_mnn PRIVATE "${MNN_ROOT}/include")
target_compile_features(karma_mnn PRIVATE cxx_std_17)
target_compile_options(karma_mnn PRIVATE -Wall -Wextra -Werror)
target_link_libraries(karma_mnn PRIVATE MNN log)
